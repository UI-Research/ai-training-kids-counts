SOLUTION (reference)

The full solution and sample prompts live in this notebook. The corrected function below (minimal, with comments) fixes the two issues: it sums only estimate columns (excludes _M) and uses the under‑18 estimate columns as the denominator. A short sample AI prompt is provided after the code.

# Prepare denominator variable list
den_vars <- vars %>%
  filter(str_detect(name, "^B17001_"),
         str_detect(label,
                    paste0("Estimate!!Total:!!(Income in the past 12 months (below|at or above) poverty level):!!(Male|Female):!!",
                           under18_ages))) %>%
    pull(name)

# Solution: corrected, minimal function
#' Calculate child poverty percent (under-18)
#' d: data.frame with ACS wide columns like B17001_XXXE
#' num_codes, den_codes: vectors of B17001 codes (no E/M suffix)
calculate_child_pov_pct_solution <- function(d, num_codes, den_codes) {
  num_cols <- paste0(num_codes, "E")
  den_cols <- paste0(den_codes, "E")

  num_present <- intersect(num_cols, names(d))
  den_present <- intersect(den_cols, names(d))

  d %>%
    transmute(
      GEOID, NAME,
      num = if (length(num_present) > 0) rowSums(across(all_of(num_present)), na.rm = TRUE) else 0,
      den = if (length(den_present) > 0) rowSums(across(all_of(den_present)), na.rm = TRUE) else 0
    ) %>%
    mutate(child_poverty_rate_pct = if_else(den > 0, 100 * num / den, NA_real_))
}

# Run the solution on the toy data
child_pov_pct_solution <- calculate_child_pov_pct_solution(county_child_pov_ar, num_vars, den_vars)

# Get Arkansas tract geometries
ar_tracts <- tracts(state = "AR", year = 2023, cb = TRUE)

# Join the cleaned poverty data from Exercise 1 solution
poverty_map_data <- ar_tracts %>%
  left_join(child_pov_pct_solution %>% select(GEOID, child_poverty_rate_pct, den), by = "GEOID")

# County-level aggregation (fast and slide-friendly)
county_map_data <- poverty_map_data %>%
  mutate(county_geoid = substr(GEOID, 1, 5)) %>%
  group_by(county_geoid) %>%
  summarize(child_poverty_rate_pct = weighted.mean(child_poverty_rate_pct, w = den, na.rm = TRUE), geometry = st_union(geometry), .groups = "drop")

p_county_solution <- ggplot(county_map_data) +
  geom_sf(aes(fill = child_poverty_rate_pct), color = "white", size = 0.2) +
  scale_fill_viridis_c(option = "magma", na.value = "grey90", labels = percent_format(accuracy = 1)) +
  labs(title = "Child Poverty by County — Arkansas", subtitle = "Weighted average of tract rates (denominator-weighted)", caption = "Toy data; replace with ACS for real results") +
  theme_void(base_size = 14) + theme(legend.position = "right", plot.title = element_text(face = "bold"))

# Save slide-ready image
ggsave("toy-data/child_poverty_county_slide_solution.png", p_county_solution, width = 10, height = 7, dpi = 300)
